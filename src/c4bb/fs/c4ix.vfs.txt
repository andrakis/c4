# C4IX VFS file listing - the manifest vfsload.c (src/c4ix/user/
# vfsload.c) reads at boot to populate C4IX's real hierarchical VFS
# (src/c4ix/vfs.c). Same format as C4KE's c4ke.vfs.txt (see that file
# for the full grammar reference); this is SOURCE, copied onto the
# disk by build-images.sh, not a build artifact.
#
# Unlike C4KE's flat ramfs, C4IX has real directories: umkdir() is
# called for every directory line as the manifest is parsed, so
# "ls /bin" and "cd" both work for real, not just as string prefixes.

# vfsload itself is deliberately excluded: it's the boot-time loader
# (built and named plainly as vfsload.c4r, not the c4ix-NAME.c4r
# convention every other user program follows), not a typical app.
BIN_ALL = hello uhello echo wc cat sh ps bench cycles ls mkdir top spin fmt

/: root
	each($BIN_ALL):
		# c4ix-hello.c4r -> /bin/hello, and an unqualified alias at the
		# root so "hello" works from the shell without the c4ix- prefix
		# or .c4r suffix
		$1: ilink(/bin/$1)
		bin/:
			$1: c4ix-$1.c4r
	usr/
		src/
			bin/
				# host filenames are prefixed c4ix-NAME.c: this disk is
				# shared with C4KE, which has its own cat.c/ps.c/top.c
				each($BIN_ALL): $1.c: c4ix-$1.c
			include/:
				c4ix.h
				c4ix_user.h
